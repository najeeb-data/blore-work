async function loadProfile() {
  const params = new URLSearchParams(window.location.search);
  const clientId = params.get('id');
  const container = document.getElementById('profile-content');

  if (!clientId) { container.innerHTML = '<h2>Business not found</h2>'; return; }

  try {
    const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
    
    // 1. Fetch Client Profile
    const profile = await getUserProfile(clientId);
    if (!profile) { container.innerHTML = '<h2>Business profile not found</h2>'; return; }

    // 2. Fetch Client's Jobs (No complex query, just filter in JS)
    const q = firestore.query(firestore.collection(db, 'jobs'), firestore.where('clientId', '==', clientId));
    const jobsSnap = await firestore.getDocs(q);
    
    let jobs = jobsSnap.docs.map(doc => ({ id: doc.id, ...doc.data() }));
    jobs = jobs.filter(j => j.status === 'open');
    jobs.sort((a, b) => (b.createdAt?.seconds || 0) - (a.createdAt?.seconds || 0));

    // 3. Setup Variables
    const logoUrl = profile.logoUrl || '/assets/icons/company-placeholder.png';
    const name = profile.businessName || 'Business';
    const rating = profile.averageRating ? profile.averageRating.toFixed(1) : 'New';
    const reviews = profile.totalReviews || 0;
    const verified = profile.isVerified ? '<i data-lucide="check-circle" style="color:#263FE5; width:20px;"></i>' : '';
    const phone = profile.phone ? profile.phone.replace(/\D/g, '') : '';
    const locationText = profile.locationArea ? `${profile.locationArea}${profile.address ? `, ${profile.address}` : ''}` : (profile.address || 'Bangalore, India');
    
    // 4. Render Jobs Grid
    let jobsHtml = '';
    if (jobs.length === 0) {
      jobsHtml = '<p class="text-muted" style="grid-column: 1/-1; text-align: center; padding: 24px;">No active jobs at the moment.</p>';
    } else {
      jobs.forEach(job => {
        jobsHtml += `
          <a href="/pages/jobs/detail.html?id=${job.id}" class="grid-card">
            <div>
              <div class="title">${job.title}</div>
              <div class="category">${job.category}</div>
            </div>
            <div>
              <div class="salary">₹${job.pay}/day</div>
              <div class="location"><i data-lucide="map-pin"></i>${job.location}</div>
            </div>
          </a>
        `;
      });
    }

    container.innerHTML = `
      <div class="card mb-4" style="border: none; background: transparent; padding: 0;">
        <div class="profile-header">
          <img src="${logoUrl}" class="profile-avatar">
          <div class="profile-info">
            <div class="profile-name">${name} ${verified}</div>
            <div class="profile-stats">
              <div><strong>${jobs.length}</strong><span>Active Jobs</span></div>
              <div><strong>${rating}</strong><span>Rating (${reviews})</span></div>
            </div>
            <div class="profile-bio">
              <div><i data-lucide="map-pin" class="icon"></i>${locationText}</div>
            </div>
          </div>
        </div>

        <div class="profile-actions">
          ${phone ? `
            <a href="https://wa.me/${phone}?text=Hi, I saw your business on BloreWork and would like to connect." target="_blank" class="btn btn-primary">
              <i data-lucide="message-circle"></i> Chat
            </a>
          ` : ''}
          <button id="share-profile-btn" class="btn btn-secondary">
            <i data-lucide="share-2"></i> Share Profile
          </button>
        </div>
      </div>

      <div style="border-top: 1px solid var(--color-border); padding-top: 24px;">
        <h3 style="margin-bottom: 16px; font-size: 16px; display: flex; align-items: center; gap: 8px;">
          <i data-lucide="briefcase"></i> Jobs Posted by ${name}
        </h3>
        <div class="jobs-grid-container">
          ${jobsHtml}
        </div>
      </div>
    `;

    if (window.lucide) window.lucide.createIcons();

    // Share Profile Logic
    document.getElementById('share-profile-btn')?.addEventListener('click', () => {
      const text = `Check out ${name} on BloreWork! They are hiring. View their profile here: ${window.location.href}`;
      window.open(`https://wa.me/?text=${encodeURIComponent(text)}`, '_blank');
    });

  } catch (err) {
    console.error(err);
    container.innerHTML = '<h2>Error loading profile</h2>';
  }
}

import { getFirebase } from '../firebase/firebase.service.js';
const SDK = 'https://www.gstatic.com/firebasejs/10.13.0';

export async function getJobs() {
  try {
    const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
    // Fetch all jobs without complex sorting to avoid index errors
    const snapshot = await firestore.getDocs(firestore.collection(db, 'jobs'));
    
    const today = new Date(); today.setHours(0, 0, 0, 0);

    return snapshot.docs
      .map(doc => ({ id: doc.id, ...doc.data() }))
      .filter(job => {
        if (job.status === 'closed') return false;
        if (!job.date) return true;
        return new Date(job.date) >= today;
      })
      .sort((a, b) => (b.createdAt?.seconds || 0) - (a.createdAt?.seconds || 0));
  } catch (err) {
    console.error("Error fetching jobs:", err);
    return []; // Return empty array instead of hanging
  }
}

export async function getJobById(jobId) {
  try {
    const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
    const docSnap = await firestore.getDoc(firestore.doc(db, 'jobs', jobId));
    return docSnap.exists() ? { id: docSnap.id, ...docSnap.data() } : null;
  } catch (err) {
    return null;
  }
}

export async function createJob(jobData) {
  const [{ db, auth }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const user = auth.currentUser;
  if (!user) throw new Error("Must be logged in to post a job.");
  const docRef = await firestore.addDoc(firestore.collection(db, 'jobs'), {
    ...jobData,
    status: 'open',
    clientId: user.uid,
    createdAt: firestore.serverTimestamp(),
    views: 0
  });
  return docRef.id;
}

export async function updateJob(jobId, jobData) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  await firestore.updateDoc(firestore.doc(db, 'jobs', jobId), jobData);
}

export async function deleteJob(jobId) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  await firestore.deleteDoc(firestore.doc(db, 'jobs', jobId));
}

export async function closeJob(jobId) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  await firestore.updateDoc(firestore.doc(db, 'jobs', jobId), { status: 'closed' });
}

import { getFirebase } from '../firebase/firebase.service.js';
const SDK = 'https://www.gstatic.com/firebasejs/10.13.0';

export async function getUserProfile(uid) {
  try {
    const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
    const docRef = firestore.doc(db, 'users', uid);
    const docSnap = await firestore.getDoc(docRef);
    return docSnap.exists() ? docSnap.data() : null;
  } catch (err) {
    console.error("Error loading profile:", err);
    return null;
  }
}

export async function saveUserProfile(uid, data) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', uid);
  await firestore.setDoc(docRef, data, { merge: true });
}

export async function toggleSavedJob(uid, jobId, isSaving) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', uid);
  await firestore.updateDoc(docRef, {
    savedJobs: isSaving ? firestore.arrayUnion(jobId) : firestore.arrayRemove(jobId)
  });
}

export async function markJobAsApplied(uid, jobId) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', uid);
  await firestore.updateDoc(docRef, { appliedJobs: firestore.arrayUnion(jobId) });
}

export async function addRecentlyViewed(uid, jobId) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', uid);
  const docSnap = await firestore.getDoc(docRef);
  if (!docSnap.exists()) return;
  let viewed = docSnap.data().recentlyViewed || [];
  viewed = viewed.filter(id => id !== jobId);
  viewed.unshift(jobId); 
  if (viewed.length > 5) viewed = viewed.slice(0, 5);
  await firestore.updateDoc(docRef, { recentlyViewed: viewed });
}

export async function completeJobWithRating(workerId, rating, jobId, reviewText, clientId, clientName) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', workerId);
  const docSnap = await firestore.getDoc(docRef);
  if (!docSnap.exists()) return;
  
  const profile = docSnap.data();
  const currentRating = profile.averageRating || 0;
  const totalReviews = profile.totalReviews || 0;
  const newTotalReviews = totalReviews + 1;
  const newAverageRating = ((currentRating * totalReviews) + rating) / newTotalReviews;
  
  let completedHistory = profile.completedJobsHistory || [];
  if (jobId && !completedHistory.includes(jobId)) completedHistory.push(jobId);
  
  await firestore.updateDoc(docRef, {
    completedJobs: firestore.increment(1),
    averageRating: newAverageRating,
    totalReviews: newTotalReviews,
    completedJobsHistory: completedHistory
  });

  if (reviewText && reviewText.trim() !== '') {
    await firestore.addDoc(firestore.collection(db, 'reviews'), {
      workerId, clientId, clientName: clientName || 'A Client', rating, text: reviewText, jobId,
      createdAt: firestore.serverTimestamp()
    });
  }
}

export async function rateClient(clientId, rating) {
  const [{ db }, firestore] = await Promise.all([getFirebase(), import(`${SDK}/firebase-firestore.js`)]);
  const docRef = firestore.doc(db, 'users', clientId);
  const docSnap = await firestore.getDoc(docRef);
  if (!docSnap.exists()) return;
  
  const profile = docSnap.data();
  const currentRating = profile.averageRating || 0;
  const totalReviews = profile.totalReviews || 0;
  const newTotalReviews = totalReviews + 1;
  const newAverageRating = ((currentRating * totalReviews) + rating) / newTotalReviews;
  
  await firestore.updateDoc(docRef, { averageRating: newAverageRating, totalReviews: newTotalReviews });
}

// Lazy-initializes Firebase only when actually needed. Uses the modular SDK from CDN —
// no bundler, no build step, per the vanilla-ESM project rule.

import { firebaseConfig, isConfigured } from './firebase-config.js';

const SDK = 'https://www.gstatic.com/firebasejs/10.13.0';
let _promise = null;

async function load() {
  if (!isConfigured()) {
    throw new Error('Firebase not configured. Edit public/js/firebase/firebase-config.js');
  }
  const [{ initializeApp }, { getAuth }, { getFirestore }] = await Promise.all([
    import(`${SDK}/firebase-app.js`),
    import(`${SDK}/firebase-auth.js`),
    import(`${SDK}/firebase-firestore.js`),
  ]);
  const app = initializeApp(firebaseConfig);
  return { app, auth: getAuth(app), db: getFirestore(app) };
}

export function getFirebase() {
  if (!_promise) _promise = load();
  return _promise;
}

    ];
  }

  function loggedInActions(user) {
    let dashRoute = '/pages/worker/dashboard.html';
    if (userRole === 'client') dashRoute = '/pages/client/dashboard.html';
    if (userRole === 'admin') dashRoute = '/pages/admin/dashboard.html';
    
    let avatarElement;
    if (userProfile && (userProfile.photoUrl || userProfile.logoUrl)) {
      avatarElement = el('img', { src: userProfile.photoUrl || userProfile.logoUrl, style: 'width: 36px; height: 36px; border-radius: 50%; object-fit: cover; border: 1px solid var(--color-border);' });
    } else {
      const initial = (user.displayName || user.email || user.phoneNumber || '?').trim()[0].toUpperCase();
      avatarElement = el('span', { style: 'display: inline-flex; align-items: center; justify-content: center; width: 36px; height: 36px; border-radius: 50%; background: var(--color-primary-soft); color: var(--color-primary); font-weight: 600; font-size: 14px;' }, initial);
    }

    return [
      themeToggle(),
      el('a', { href: dashRoute, title: 'Go to Dashboard', style: 'text-decoration: none;' }, avatarElement),
      el('button', { class: 'btn btn-secondary btn-sm', style: 'padding:0; width:36px; height:36px; border:none; background:transparent; display:flex; align-items:center; justify-content:center; color: var(--color-text-muted);', type: 'button', onClick: async () => { await signOut(); } }, [ 
        el('i', { 'data-lucide': 'log-out', style: 'width:20px;' }) 
      ])
    ];
  }

  function render() {
    clearChildren(root);
    const actions = currentUser ? loggedInActions(currentUser) : [themeToggle(), ...loggedOutActions()];

    root.append(
      el('div', { class: 'navbar' }, [
        el('div', { class: 'navbar__inner' }, [
          el('a', { class: 'navbar__logo', href: ROUTES.HOME }, [
            el('img', { src: '/assets/logo.png', alt: 'BloreWork Logo', class: 'navbar__logo-img' }),
            el('span', { class: 'navbar__logo-text' }, APP_NAME)
          ]),
          el('nav', { class: 'navbar__links' }, [
            el('a', { class: 'navbar__link', href: '/pages/jobs/index.html' }, 'Find Jobs'),
          ]),
          el('div', { class: 'navbar__actions' }, actions),
        ]),
      ])
    );
    if (window.lucide) window.lucide.createIcons();
  }

  render();
  await onAuthChange(async (user) => { 
    currentUser = user; 
    if (user) {
      try {
        userProfile = await getUserProfile(user.uid);
        if (userProfile && userProfile.role) userRole = userProfile.role;
      } catch (err) {}
    } else {
      userProfile = null;
    }
    render(); 
  });
}

import { el, qs, clearChildren } from '../utils/dom-utils.js';
import { ROUTES } from '../config/app-config.js';
import { onAuthChange, getCurrentUserRole } from '../firebase/auth.service.js';

export async function initBottomNav() {
  const root = qs('#bottom-nav-root');
  if (!root) return;

  async function render(user) {
    clearChildren(root);
    document.body.classList.toggle('has-bottom-nav', !!user);
    if (!user) return;

    const role = await getCurrentUserRole();
    const currentPath = window.location.pathname;

    let dashRoute = '/pages/worker/dashboard.html';
    if (role === 'client') dashRoute = '/pages/client/dashboard.html';
    if (role === 'admin') dashRoute = ROUTES.ADMIN_DASHBOARD;

    let profileRoute = '/pages/worker/profile.html';
    if (role === 'client') profileRoute = '/pages/client/profile.html';

    const items = [
      { label: 'Home', icon: 'home', href: ROUTES.HOME, match: '/' },
      { label: 'Search', icon: 'search', href: ROUTES.JOBS, match: '/pages/jobs/index.html' },
      role === 'client'
        ? { label: 'Post', icon: 'plus-circle', href: ROUTES.POST_JOB, match: '/pages/client/post-job.html' }
        : { label: 'Saved', icon: 'bookmark', href: '/pages/worker/saved.html', match: '/pages/worker/saved.html' },
      { label: 'Account', icon: 'user', href: profileRoute, match: profileRoute },
    ];

    root.append(
      el('div', { class: 'bottom-nav' }, [
        el('div', { class: 'bottom-nav__inner' },
          items.map((it) => {
            const isActive = currentPath === it.match || (it.match === '/' && currentPath === '/index.html');
            return el('a', { class: `bottom-nav__link ${isActive ? 'active' : ''}`, href: it.href }, [
              el('i', { 'data-lucide': it.icon, style: 'width: 20px; height: 20px;' }),
              el('span', {}, it.label)
            ]);
          })
        ),
      ])
    );
    if (window.lucide) window.lucide.createIcons();
  }

  render(null);
  await onAuthChange(render);
}
